int nums[N];
void quick_sort(int l , int r)
{
    if(l>=r)  return;
    int i = l-1 , j = r+1 , x = nums[l+r>>1];
    while(i<j)
    {
        //do...while() 防止i、j都遇到枢轴量陷入死循环
        do i++; while(nums[i]<x);
        do j--; while(nums[i]>x);

### 归并排序

```cpp
int nums[N] , temp[N];
void merge_sort(int l , int r)
{
    if(l>=r)  return;
    int mid = l+r >> 1;
    merge_sort(l,mid) , merge_sort(mid+1,r);
    int i = l , j = mid+1 , k=0;
    while(i<=mid&&j<=r)
    {
        if(nums[i]<nums[j])  temp[k++] = nums[i++];
        else  temp[k++]=nums[j++];
    }
    while(i<=mid)  temp[k++]=nums[i++];
    while(j<=r)  temp[k++]=nums[j++];
    for(int i=0,j=l;i<k;i++,j++)  nums[j]=temp[i];
}
```

## 二分

### 整数二分

```cpp
/*查找左区间的右边界*/
//check(i)  当下标i在左区间时返回true  在右区间时返回false
int nums[N];
int b_search(int l , int r)
{
    //每次二分区间，同时确保答案在新的区间内
    while(l<r)
    {
        int mid = l+r+1 >> 1;   //l+r+1防止陷入死循环
        if(check(mid))  l = mid;
        else  r = mid-1;
    }
    return l;
}


/*查找右区间的左边界*/
//check(i)  当下标i在左区间时返回false  在右区间时返回true
int b_search(int l , int r)
{
    while(l<r)
    {
        int mid = l+r >> 1;  //l+r不用取1防止陷入死循环
        if(check(mid))  r = mid;
        else  l = mid+1;
    }
    return l;
}
```

> * 当查找的元素不存在时，此时函数会返回边界，务必**特殊处理**

vector<int,int> segs;   //存放若干个存在重叠的区间
void merge(vector<int,int>& segs)
{
    sort(segs.begin() , segs.end());
    int st = -INF , ed = -INF;
    vector<int,int> res;
    for(auto& ele : segs)
    {
        if(ele.first>ed)
        {
            if(st!=-INF)  res.push_back({st,ed});
            st = ele.first , ed = ele.second;
        }
        else
        {
            ed = max(ed , ele.second);

    segs = res;
}

### 中缀表达式求值

**中缀表达式为表达式树的中序遍历，求值过程则是模拟中序遍历的过程中完成计算**

```cpp
//当一颗子树全部遍历完时可进行计算 -> 当新的算符<上一个算符时，子树已经遍历完并且开始向根的方向遍历
#include<stack>
#include<map>
using namespace std;

unordered_map<char,int>  pr = {{'+',1},{'-',1},{'*',2},{'/',2}};  //存储算符优先级
stack<char> op;  //存储操作符
stack<int> nums;  //存储操作数

//nums中取两个操作数，op中取一个操作符进行运算并将结果压回栈内
void eval()
{
    int b = nums.top() , nums.pop();
    int a = nums.top() , nums.pop();
    char c = op.top() , op.pop();
    if(c=='+')   nums.push(a+b);
    else if(c=='-')  nums.push(a-b);
    else if(c=='*')  nums.push(a*b);
    else  nums.push(a/b);
}

int main()
{
    string str;  cin >> str;
    for(int i=0;i<str.length();i++)
    {
        char c = str[i];
        if(isdigit(c))
        {
            int val = 0 , j = i;
            //双指针取下数字本体
            while(j<str.length() && isdigit(str[j]))
            {
                val = val*10+str[j]-'0';
            }
            i = j-1;   //同步i
            nums.push(val);
        }
        else if(c=='(')  op.push(c);
        else if(c==')')  
        {
            while(op.top()!='(')  eval();
            op.pop();  //弹出左括号
        }
        else
        {
            while(op.size() && pr[op.top()]>=pr[c])  eval();
            op.push(c);
        }
    }
    while(op.size())  eval();
    cout << nums.top() << endl;
    return 0;
}
```

### Trie树

**采用树的方式存储字符串，使得不同字符串的相同前缀存储一次，实现了字符串的快速存储和查找**

//trie[][]存储trie树，cnt[i]存储以第i个结点结尾的字符串个数,idx指向第一个空位置
int trie[n][m] , cnt[n] , idx;   //n表示结点总个数,m表示分支总个数

//将字符串s存入trie树

{
    int i = 0;   //第0行存储根节点

### 并查集

**逻辑结构上是一种集合，常用于合并两个集合以及判断两个元素是否属于同一集合，故采用树来存储**

> 此处的树则又采用双亲表示法，方便寻根，用每棵树的根来标识一个集合

```cpp
int ufs[n];

//初始化
void init()
{
    for(int i=0;i<n;i++)  ufs[i] = -1;   //每个结点都是自身所在集合的根结点
}

//查找元素x所在集合的根结点+压缩路径
int find(int x)
{
    if(ufs[x]<0)  return x;
    return ufs[x] = find(ufs[x]);
}

//合并元素x,y所在的集合
void merge(int x , int y)
{
    int rx = find(x) ,  ry = find(y);
    if(rx==ry)  return;
    if(ufs[rx]>ufs[ry])  
    {
        ufs[ry]+=ufs[rx];
        ufs[rx] = ry;
    }
    else
    {
        ufs[rx]+=ufs[ry];
        ufs[ry] = rx;
    }
}
```





### 堆

**小顶堆，本质上逻辑结构是一颗完全二叉树，同时满足每个节点小于左、右子树上的所有结点**

> 完全二叉树采用顺序存储

```cpp
int h[n] , idx , len;  //h[]存储堆 ， idx指向第一个空位置 ， len记录堆中元素个数

void init()
{
    idx = 1; //从1开始存放，方便利用完全二叉树父节点、子节点间的序号关系
}

//堆的核心操作都可以拆解为up()、down()操作的组合

//将序号为x的的元素向上移动至正确位置
void up(int x)
{
    while(x/2 && h[x/2]>h[x])  
    {
        swap(h[x/2],h[x]);
        x >>= 2;
    }
}

//将下标为x的元素下移至正确位置
void down(int x)
{
    int temp = x;
    if(2*x<=len && h[2*x]<h[temp])  temp = 2*x;
    if(2*x+1<=len && h[2*x+1]<h[temp])  temp = 2*x+1;
    if(x!=temp)
    {
        swap(h[x],h[temp]);
        down(temp);
    }
}

//堆的初始化_1  复杂度为 O(nlogn)
int nums[n];
for(int i=0;i<n;i++)
{
    h[idx] = nums[i];
    len++;
    up(idx++);
}
//堆的初始化_2  复杂度为 O(n)
for(int i=0;i<n;i++)  h[i+1] = nums[i];
for(int i=x/2;i>0;i--)  down(i);

//删除堆顶元素
h[1] = h[len--];
down(1);
```





### 哈希

**哈希的核心思想为映射，构建值到下标，下标到下标的映射关系**

```cpp
//哈希通过哈希函数将大范围数据映射到小范围，此时会面临不同值通过hash函数映射到同一位置的情况，称之为冲突
//解决冲突的常见方法为：拉链法 、 开放定址法

/*拉链法*/
const int N = 1e5+3;  //通常取大于期望范围的最小质数，数学原理上能减少冲突
int h[N] , e[N] ,ne[N] , idx;

void init()
{
    for(int i=0;i<N;i++)  h[i] = -1;  //初始状况下每个链表都为空
}

void insert(int x)
{
    int t = ((x%N)+N)%N;
    //链表头插法
    e[idx] = x;
    ne[idx] = h[t];
    h[t] = idx++;
}

bool find(int x)
{
    int t = ((x%N)+N)%N;
    for(int i=h[t];i!=-1;i=ne[i])
    {
        if(e[i]==x)  return true;
    }
    return false;
}


/*开放定址法*/
const int N = 2e5+3;   //通常取实际范围的2~3倍，有效减少冲突
const int null = 0x3f3f3f3f;
int h[N];

//将表中每个元素设置为一个题中数据取不到的值，用来标识没有数据存储
void init()
{
    for(int i=0;i<N;i++)  h[i] = null;
}

//若x在表中，则返回存储下标 ； 如x不在表中，则返回应该存储的位置
int find(int x)
{
    int t = ((x%N)+N)%N;
    while(h[t]!=null && h[t]!=x)  
    {
        t++;
        if(t==N)  t=0;
    }
    return t;
}
```





### 字符串哈希

**定义字符串哈希函数，使得每个字符串对应的哈希值唯一，同时预处理字符串前缀哈希数组，方便快速求出字符串中任一一个字符串的哈希值，并比较异同**

```cpp
//字符串的hash函数：将字符串视为p进制数，然后求出数值大小后 mod R
//当 p=131 /13331 , r=2^64 时，每个字符串的哈希值可以认为唯一

//涉及到一些取模运算性质
const int p = 131;
typedef unsigned long long ull;  //采用unsigned long long，数据自动取模2^64
ull pre[n] , temp[n];   //temp[i]存储p^i避免反复计算
string s;
temp[0]=1;

//pre[]的构建
for(int i=1;i<=s.length();i++)
{
    temp[i] = temp[i-1]*p;
    pre[i] = pre[i-1]*p+s[i-1];
}

//pre[]的使用：求下标为l~r的子串的哈希值
ull res = pre[r]-pre[l-1]*temp[r-l+1];
```







## KMP模式串匹配

### KMP_1

**next[i]标识第i位字符失配时跳转到next[i]位继续比较**

```cpp
//求解next[]数组
int next[n];
string s;
void get_next()
{
    int i=1 , j=0;
    next[i] = j;
    while(i<s.length()-1)
    {
        if(!j || s[i]==s[j])
        {
            i++ , j++;
            //next[i] = j;    可以进一步优化
            if(s[i]==s[j])  next[i] = next[j];
            else  next[i] = j;
        }
        else
        {
            j = next[j];
        }
    }
}

//模式串匹配 : 在str中查找子串t的下标
int find(string str , string t)
{
    int i=0 , j=0;
    while(i<str.length() && j<t.length())
    {
        if(!j || str[i]==t[j])
        {
            i++ , j++;
        }
        else
        {
            j = next[j];
        }
    }
    if(j==t.length())
    {
        return i-j;
    }
    else
    {
        return -1;
    }
}
```

int next[n];
