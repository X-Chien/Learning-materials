```cpp
void quick_sort(int l , int r)
{
    if(l>=r)  return;
    int i = l-1 , j = r+1 , x = nums[l+r>>1];
    while(i<j)
    {
```

### 归并排序

```cpp
int nums[N] , temp[N];
void merge_sort(int l , int r)
{
    if(l>=r)  return;
    int mid = l+r >> 1;
    merge_sort(l,mid) , merge_sort(mid+1,r);
    int i = l , j = mid+1 , k=0;
    while(i<=mid&&j<=r)
    {
        if(nums[i]<nums[j])  temp[k++] = nums[i++];
        else  temp[k++]=nums[j++];
    }
    while(i<=mid)  temp[k++]=nums[i++];
    while(j<=r)  temp[k++]=nums[j++];
    for(int i=0,j=l;i<k;i++,j++)  nums[j]=temp[i];
}
```

## 二分

### 整数二分

```cpp
/*查找左区间的右边界*/
//check(i)  当下标i在左区间时返回true  在右区间时返回false
int nums[N];
int b_search(int l , int r)
{
    //每次二分区间，同时确保答案在新的区间内
    while(l<r)
    {
        int mid = l+r+1 >> 1;   //l+r+1防止陷入死循环
        if(check(mid))  l = mid;
        else  r = mid-1;
    }
    return l;
}


/*查找右区间的左边界*/
//check(i)  当下标i在左区间时返回false  在右区间时返回true
int b_search(int l , int r)
{
    while(l<r)
    {
        int mid = l+r >> 1;  //l+r不用取1防止陷入死循环
        if(check(mid))  r = mid;
        else  l = mid+1;
    }
    return l;
}
```

> * 当查找的元素不存在时，此时函数会返回边界，务必**特殊处理**

```cpp
vector<int,int> segs;   //存放若干个存在重叠的区间
void merge(vector<int,int>& segs)
{
    sort(segs.begin() , segs.end());
    int st = -INF , ed = -INF;
    vector<int,int> res;
    for(auto& ele : segs)
    {
        if(ele.first>ed)
        {
            if(st!=-INF)  res.push_back({st,ed});
            st = ele.first , ed = ele.second;

        else
        {

}
```

### 中缀表达式求值

**中缀表达式为表达式树的中序遍历，求值过程则是模拟中序遍历的过程中完成计算**

```cpp
//当一颗子树全部遍历完时可进行计算 -> 当新的算符<上一个算符时，子树已经遍历完并且开始向根的方向遍历
#include<stack>
#include<map>
using namespace std;

unordered_map<char,int>  pr = {{'+',1},{'-',1},{'*',2},{'/',2}};  //存储算符优先级
stack<char> op;  //存储操作符
stack<int> nums;  //存储操作数

//nums中取两个操作数，op中取一个操作符进行运算并将结果压回栈内
void eval()
{
    int b = nums.top() , nums.pop();
    int a = nums.top() , nums.pop();
    char c = op.top() , op.pop();
    if(c=='+')   nums.push(a+b);
    else if(c=='-')  nums.push(a-b);
    else if(c=='*')  nums.push(a*b);
    else  nums.push(a/b);
}

int main()
{
    string str;  cin >> str;
    for(int i=0;i<str.length();i++)
    {
        char c = str[i];
        if(isdigit(c))
        {
            int val = 0 , j = i;
            //双指针取下数字本体
            while(j<str.length() && isdigit(str[j]))
            {
                val = val*10+str[j]-'0';
            }
            i = j-1;   //同步i
            nums.push(val);
        }
        else if(c=='(')  op.push(c);
        else if(c==')')  
        {
            while(op.top()!='(')  eval();
            op.pop();  //弹出左括号
        }
        else
        {
            while(op.size() && pr[op.top()]>=pr[c])  eval();
            op.push(c);
        }
    }
    while(op.size())  eval();
    cout << nums.top() << endl;
    return 0;
}
```

### Trie树

```cpp
//trie[][]存储trie树，cnt[i]存储以第i个结点结尾的字符串个数,idx指向第一个空位置
int trie[n][m] , cnt[n] , idx;   //n表示结点总个数,m表示分支总个数

//将字符串s存入trie树

{
```

### 并查集

**逻辑结构上是一种集合，常用于合并两个集合以及判断两个元素是否属于同一集合，故采用树来存储**

> 此处的树则又采用双亲表示法，方便寻根，用每棵树的根来标识一个集合

```cpp
int ufs[n];

//初始化
void init()
{
    for(int i=0;i<n;i++)  ufs[i] = -1;   //每个结点都是自身所在集合的根结点
}

//查找元素x所在集合的根结点+压缩路径
int find(int x)
{
    if(ufs[x]<0)  return x;
    return ufs[x] = find(ufs[x]);
}

//合并元素x,y所在的集合
void merge(int x , int y)
{
    int rx = find(x) ,  ry = find(y);
    if(rx==ry)  return;
    if(ufs[rx]>ufs[ry])  
    {
        ufs[ry]+=ufs[rx];
        ufs[rx] = ry;
    }
    else
    {
        ufs[rx]+=ufs[ry];
        ufs[ry] = rx;
    }
}
```

**小顶堆，本质上逻辑结构是一颗完全二叉树，同时满足每个节点小于左、右子树上的所有结点**

> 完全二叉树采用顺序存储

```cpp
int h[n] , idx , len;  //h[]存储堆 ， idx指向第一个空位置 ， len记录堆中元素个数

void init()
{
    idx = 1; //从1开始存放，方便利用完全二叉树父节点、子节点间的序号关系
}

//堆的核心操作都可以拆解为up()、down()操作的组合

//将序号为x的的元素向上移动至正确位置
void up(int x)
{
    while(x/2 && h[x/2]>h[x])  
    {
        swap(h[x/2],h[x]);
        x >>= 2;
    }
}

//将下标为x的元素下移至正确位置
void down(int x)
{
    int temp = x;
    if(2*x<=len && h[2*x]<h[temp])  temp = 2*x;
    if(2*x+1<=len && h[2*x+1]<h[temp])  temp = 2*x+1;
    if(x!=temp)
    {
        swap(h[x],h[temp]);
        down(temp);
    }
}

//堆的初始化_1  复杂度为 O(nlogn)
int nums[n];
for(int i=0;i<n;i++)
{
    h[idx] = nums[i];
    len++;
    up(idx++);
}
//堆的初始化_2  复杂度为 O(n)
for(int i=0;i<n;i++)  h[i+1] = nums[i];
for(int i=x/2;i>0;i--)  down(i);

//删除堆顶元素
h[1] = h[len--];
down(1);
```

### DFS

> **常用于解决排列问题，核心在于找到排列思路，以及排列过程中的剪枝优化和恢复现场**

### 树与图的深度优先遍历

```cpp
#include<cstring>
#include<algorithm>
using namespace std;

const int inf = 0x3f3f3f3f;
int h[N] , e[M] , ne[M] , idx;

void add(int a , int b)
{
    e[idx] = b;
    ne[idx] = h[a];
    h[a] = idx++;

int DFS(int x)     
{
    st[x] = 1;
    int sum = 1;  //存储以当前结点为根结点的子树结点个数
    int size = 0;  //存储去除当前节点后各联通分支结点个数最大值

        int j = e[i];
        if(!st[j])
        {
            int t = DFS(j);
            sum += t;
            size = max(size , t);
        }
    }
    size = max(n-sum , size);
    res = min(res , size);
    return sum;
}

int main()
{

    cin >> n;
    for(int i=0;i<n-1;i++)
    {
        int a , b;  cin >> a >> b;
        add(a , b) , add(b , a);
    }
    DFS(1);
    cout << res << endl;
    return 0;
}
```

### Dijkstra单源最短路

```cpp
/*朴素版Dijkstra*/
//复杂都为O(n^2) ， 适用于稠密图
#include<iostream>
#include<cstring>
#include<algorithm>

using namespace std;

const int N = 510 , M = 1e5+10;
const int inf = 0x3f3f3f3f;
int g[N][N];
int n , m;
int dist[N];   //每个顶点到起点的距离
bool st[N];    //判断结点是否已经找到最短距离

void Dijkstra()
{
    memset(dist,0x3f,sizeof dist);
    dist[1] = 0;
    for(int i=0;i<n-1;i++)   //计数循环
    {
        int t = -1;
        for(int j=1;j<=n;j++)
        {
            if(!st[j] && (t==-1||dist[j]<dist[t]))  t = j;
        }
        st[t] = 1;
        for(int i=1;i<=n;i++)
        {
            dist[i] = min(dist[i] , dist[t]+g[t][i]);
        }
    }
    if(dist[n]==inf)  cout << -1 << endl;
    else  cout << dist[n] << endl;
}

int main()
{
    cin >> n >> m;
    memset(g,0x3f,sizeof g);
    while(m--)
    {
        int a , b , w;  cin >> a >> b >> w;
        g[a][b] = min(g[a][b] , w);
    }
    Dijkstra();
    return 0;
}

#include<queue>

using namespace std;
typedef pair<int,int> PII;

const int N = 2e5+10;
const int inf = 0x3f3f3f3f;
int h[N] , e[N] , ne[N] , w[N] , idx;
priority_queue<PII,vector<PII>,greater<PII>> heap;  //小顶堆自动维护当前距离的最小值
int dist[N];
bool st[N];
int n , m;

void add(int a , int b , int c)
{
    e[idx] = b;
    ne[idx] = h[a];
    h[a] = idx;
    w[idx] = c;
    idx++;
}

void Dijkstra()
{
    memset(dist,0x3f,sizeof dist);
    dist[1] = 0;
    heap.push({0,1});   //距离在前，序号在后 ， 实现按边长排序
    while(heap.size())
    {
        auto t = heap.top();
        heap.pop();  //取堆顶
        int distance = t.first , x = t.second;
        if(st[x])  continue;  //跳过冗余边，减少时间浪费
        st[x] = 1;
        //更新每个点到起点的距离
        for(int i=h[x];i!=-1;i=ne[i])
        {
            int j = e[i];
            if(dist[j] > dist[x]+w[i])
            {
                dist[j] = dist[x]+w[i];
                heap.push({dist[j],j});
            }
        }
    }
    if(dist[n]==inf)  cout << -1 << endl;
    else  cout << dist[n] << endl;
}

int main()
{
    memset(h,-1,sizeof h);
    cin >> n >> m;
    while(m--)
    {
        int a , b , w;  cin >> a >> b >> w;
        add(a,b,w);
    }
    Dijkstra();
    return 0;
}
```

### Bellman_Ford单源最短路

**通过(n-1)次对所有边的循环，第k次循环保证最短路径边数<=k的结点全部被找出**

> 适用于含有负权边的图

```cpp
#include<iostream>
#include<algorithm>
#include<cstring>

using namespace std;
const int inf = 0x3f3f3f3f;
const int N = 510 , M = 1e5+10;
struct Edge
{
    int a , b , w;
}edge[M];
int dist[N] , backup[N];
int n , m , k;

void Bellman_Ford()
{
    memset(dist,0x3f,sizeof dist);
    dist[1] = 0;  //初始化起点距离
    for(int i=0;i<k;i++)   //计数循环
    {
        //进行备份，防止串联
        memcpy(backup , dist , sizeof dist);
        for(int i=0;i<m;i++)
        {
            int a = edge[i].a , b = edge[i].b , w = edge[i].w;
            dist[b] = min(dist[b] , backup[a]+w);  //松弛
        }
    }
    if(dist[n] > inf/2)  cout << "impossible" << endl;
    else  cout << dist[n] << endl;
}

int main()
{
    cin >> n >> m >> k;
    for(int i=0;i<m;i++)  cin >> edge[i].a >> edge[i].b >> edge[i].w;
    Bellman_Ford();
    return 0;
}
```

### SPFA单源最短路

**SPFA本质上是借助队列优化了Bellman——Ford算法，省去了很多不必要的松弛操作**

```cpp
#include<iostream>
#include<queue>
#include<cstring>

using namespace std;

const int N = 1e5+10;
const int inf = 0x3f3f3f3f;
int h[N] , e[N] , ne[N] , w[N] , idx;
queue<int> q;
int dist[N];
bool st[N];  //记录结点是否已经在队列中，避免重复入队
int n , m; 
void add(int a , int b , int c)
{
    e[idx] = b;
    ne[idx] = h[a];
    h[a] = idx;
    w[idx] = c;
    idx++;
}

void SPFA()
{
    memset(dist , 0x3f , sizeof dist);
    dist[1] = 0;
    st[1] = 1;
    q.push(1);
    //借助队列一层一层松弛
    while(q.size())
    {
        auto t = q.front();
        q.pop();
        st[t] = 0;   //元素已经出队列
        for(int i=h[t];i!=-1;i=ne[i])
        {
            int j = e[i];
            if(dist[j] > dist[t]+w[i])
            {
                dist[j] = dist[t]+w[i];  //松弛
                if(!st[j])  
                {
                    q.push(j);
                    st[j] = 1;  //剪枝
                }
            }
        }
    }
    if(dist[n]==inf)  cout << "impossible" << endl;
    else  cout << dist[n] << endl;
}

int main()
{
    memset(h,-1,sizeof h);
    cin >> n >> m;
    while(m--)
    {
        int a , b , w;  cin >> a >> b >> w;
        add(a,b,w);
    }
    SPFA();
    return 0;
}
```





### Floyd多源最短路

```cpp
#include<iostream>
#include<algorithm>
#include<cstring>

using namespace std;

const int N = 210;
const int inf = 0x3f3f3f3f;
int g[N][N];
int n , m , k;

void Floyd()
{
    for(int k=1;k<=n;k++)   //枚举每个顶点
    {
        for(int i=1;i<=n;i++)
        {
            for(int j=1;j<=n;j++)
            {
                g[i][j] = min(g[i][j] , g[i][k]+g[k][j]);
            }
        }
    }
}

int main()
{
    cin >> n >> m >> k;
    for(int i=1;i<=n;i++)
        for(int j=1;j<=n;j++)
            if(i==j)  g[i][j]=0;
            else  g[i][j] = inf;
    while(m--)
    {
        int a , b , w;  cin >> a >> b >> w;
        g[a][b] = min(g[a][b] , w);
    }
    Floyd();
    while(k--)
    {
        int a , b;  cin >> a >> b;
        if(g[a][b] > inf/2)  cout << "impossible" << endl;
        else  cout << g[a][b] << endl;
    }
    return 0;
}
```

## 最小生成树

### Prim算法

```cpp
#include<iostream>
#include<algorithm>
#include<cstring>

using namespace std;
const int N = 510;
const int inf = 0x3f3f3f3f;
int g[N][N];
int n , m;
int dist[N] , st[N];
int res;

void Prim()
{
    
    memset(dist , 0x3f , sizeof dist);
    for(int i=0;i<n;i++)
    {
        int t = -1;
        for(int j=1;j<=n;j++)
        {
            if(!st[j] && (t==-1||dist[j]<dist[t]))  t = j;  //找到没有纳入生成树，且到生成树距离最小的点
        }
        if(i && dist[t]==inf)
        {
            cout << "impossible" << endl;
            return;
        }
        st[t] = 1;
        if(i)  res += dist[t];
        //更新距离
        for(int i=1;i<=n;i++)
        {
            dist[i] = min(dist[i] , g[t][i]);
        }
    }
    cout << res << endl;
}

int main()
{
    cin >> n >> m;
    for(int i=1;i<=n;i++)
    {
        for(int j=1;j<=n;j++)
        {
            if(i==j)  g[i][j] = 0;
            else  g[i][j] = inf;
        }
    }
    while(m--)
    {
        int a , b , w;  cin >> a >> b >> w;
        g[a][b] = g[b][a] = min(g[a][b] , w);
    }
    Prim();
    return 0;
}
```

### Kruskal算法

```cpp
#include<iostream>
#include<algorithm>
#include<cstring>

using namespace std;
const int N = 2e5+10;
struct Edge
{
    int a , b , w;
    bool operator< (const Edge& e)  //重载Edge的＜运算，方便排序
    {
        return w < e.w;
    }
}edge[N];
int n , m;
int res , cnt;
int p[N];  //并查集判断待选边是否造成回路

int find(int x)
{
    if(p[x]<0) return x;
    else  return p[x] = find(p[x]);
}

void Kruskal()
{
    sort(edge , edge+m);   //按权值排序
    for(int i=0;i<m;i++)
    {
        int a = edge[i].a , b = edge[i].b;
        a = find(a) , b = find(b);
        if(a==b)  continue;
        res += edge[i].w;
        cnt++;
        p[a] = b;
    }
    if(cnt < n-1)  cout << "impossible" << endl;
    else  cout << res << endl;
}

int main()
{
    cin >> n >> m;
    for(int i=1;i<=n;i++)  p[i] = -1;
    for(int i=0;i<m;i++)  cin >> edge[i].a >> edge[i].b >> edge[i].w;
    Kruskal();
    return 0;
}
```

```cpp
/*染色法+DFS*/
#include<iostream>
#include<cstring>

using namespace std;
const int N = 1e5+10 , M = 2*N;
int color[N];
int h[N] , e[M] , ne[M] , idx;
int n , m;

void add(int a , int b)
{
    e[idx] = b;
    ne[idx] = h[a];
    h[a] = idx++;
}

bool DFS(int x , int c)
{
    color[x] = c;
    for(int i=h[x];i!=-1;i=ne[i])
    {
        int j = e[i];
        if(!color[j])  //还没有染色
        {
            if(!DFS(j,3-c))  return 0;
        }
        else
        {
            if(color[j]==color[x])  return 0;
        }
    }
    return 1;
}

int main()
{
    memset(h,-1,sizeof h);
    cin >> n >> m;
    while(m--)
    {
        int a , b;  cin >> a >> b;
        add(a,b) , add(b,a);
    }
    for(int i=1;i<=n;i++)
    {
        if(!color[i])
        {
            if(!DFS(i,1))  
            {
                cout << "No" << endl;
                return 0;
            }
        }
    }
    cout << "Yes" << endl;
    return 0;
}
```

### 二分图的最大匹配

```cpp
#include<iostream>
#include<cstring>

using namespace std;

const int N = 510;
int n1 , n2 , m;
int g[N][N];   //记录左半部分结点与右半部分结点间是否有边
int match[N];  //记录与右半部分结点匹配的左半部分结点
int st[N];  //记录状态
int res;

//帮左半部分x号结点匹配
bool find(int x)
{
    for(int i=1;i<=n2;i++)
    {
        if(g[x][i] && !st[i])
        {
            st[i] = 1;
            if(!match[i] || find(match[i]))
            {
                match[i] = x;
                return 1;
            }
        }
    }
    return 0;  //最坏的情况为某次递归时，该结点只有一条边，无法进行协调
}

int main()
{
    cin >> n1 >> n2 >> m;
    while(m--)
    {
        int a , b;  cin >> a >> b;
        g[a][b] = 1;
    }
    for(int i=1;i<=n1;i++)
    {
        memset(st,0,sizeof st);
        if(find(i))  res++;
    }
    cout << res << endl;
    return 0;
}
```
